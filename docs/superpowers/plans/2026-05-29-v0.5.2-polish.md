等等，Goldmark 的扩展机制比较复杂。让我换一种更简单的方法。

行高亮的更简单方法：使用两阶段处理。在 markdown 渲染前，扫描源码找出所有带 `{hl_lines=...}` 的 fenced code block，记录位置和行号信息，从源码中移除 `{hl_lines=...}` 标注。渲染后，按顺序给 `<pre class="chroma">` 注入 `data-hl-lines` 属性。

在 `internal/content/renderer.go` 中新增方法：

```go
import "regexp"

// hlLinesRe matches {hl_lines=[...]} in fenced code info strings.
var hlLinesRe = regexp.MustCompile(`\{hl_lines=\[([0-9,\-\s]+)\]\}`)

// extractHLLines extracts and returns hl_lines annotations from markdown source,
// and returns the cleaned source without these annotations.
func extractHLLines(source []byte) (cleaned []byte, hlInfo []string) {
	hlInfo = []string{}
	cleaned = hlLinesRe.ReplaceAll(source, []byte(""))
	matches := hlLinesRe.FindAllSubmatch(source, -1)
	for _, m := range matches {
		hlInfo = append(hlInfo, string(m[1]))
	}
	return
}

// injectHLLines injects data-hl-lines attributes into <pre class="chroma"> elements.
func injectHLLines(html string, hlInfo []string) string {
	if len(hlInfo) == 0 {
		return html
	}
	preRe := regexp.MustCompile(`<pre class="chroma">`)
	idx := 0
	return preRe.ReplaceAllStringFunc(html, func(match string) string {
		if idx < len(hlInfo) && hlInfo[idx] != "" {
			result := fmt.Sprintf(`<pre class="chroma" data-hl-lines="%s">`, hlInfo[idx])
			idx++
			return result
		}
		idx++
		return match
	})
}
```

然后修改 `RenderHTML`:

```go
func (r *Renderer) RenderHTML(source []byte) (string, error) {
	cleaned, hlInfo := extractHLLines(source)
	var buf bytes.Buffer
	if err := r.md.Convert(cleaned, &buf); err != nil {
		return "", fmt.Errorf("render markdown: %w", err)
	}
	return injectHLLines(buf.String(), hlInfo), nil
}
```

```js
/* ================================================================ */
/*  3.5. Highlight Lines                                             */
/* ================================================================ */

function initHighlightLines() {
  document.querySelectorAll('pre.chroma[data-hl-lines]').forEach(function (pre) {
    var ranges = pre.getAttribute('data-hl-lines');
    if (!ranges) return;
    var hlSet = new Set();
    ranges.split(',').forEach(function (part) {
      part = part.trim();
      if (part.includes('-')) {
        var segs = part.split('-');
        var start = parseInt(segs[0], 10);
        var end = parseInt(segs[1], 10);
        for (var i = start; i <= end; i++) hlSet.add(i);
      } else {
        hlSet.add(parseInt(part, 10));
      }
    });

    // Each <span class="line"> contains one line; highlight matching ones
    var lines = pre.querySelectorAll('.line');
    lines.forEach(function (line, i) {
      if (hlSet.has(i + 1)) {
        line.classList.add('hl');
      }
    });
  });
}
```

## Task 3: 归档页月份倒序排列

**Files:**
- Modify: `internal/index/site.go:22-25` (Archive struct)
- Modify: `internal/index/site.go:80-87` (BuildSite archive population)
- Modify: `themes/zhenhai/layouts/list.html:9-38` (archive template)

### 问题分析

当前 `Archive.Items` 是 `map[int]map[time.Month][]*content.Page`，Go 的 map 迭代顺序是随机的，因此模板中 `{{range $month, $monthPages := $months}}` 的月份顺序不可预测。需要改为有序结构 `[]ArchiveMonth`，并按 12→1 倒序排列。

### 修改

- [ ] **Step 1: 修改 Archive 数据结构**

修改 `internal/index/site.go`:

```go
// ArchiveMonth holds pages for a specific month.
type ArchiveMonth struct {
	Month time.Month
	Pages []*content.Page
}

// Archive holds pages grouped by year then month (months sorted 12→1).
type Archive struct {
	Years []int
	Items map[int][]ArchiveMonth
}
```

```go
// appendArchiveMonth appends a page to the correct month slice, creating
// a new ArchiveMonth entry when needed.
func appendArchiveMonth(months []ArchiveMonth, month time.Month, page *content.Page) []ArchiveMonth {
	for i := range months {
		if months[i].Month == month {
			months[i].Pages = append(months[i].Pages, page)
			return months
		}
	}
	// New month
	found := false
	for _, y := range months {
		if y.Month == month {
			found = true
			break
		}
	}
	_ = found
	return append(months, ArchiveMonth{Month: month, Pages: []*content.Page{page}})
}
```
